\label{ch:jt}

\begin{usesbox}
\textbf{This chapter uses:} the spherical reduction of the action (\cref{ch:reduction}); the throat in synchronous coordinates (\cref{thm:throatevo}); the junction position $\sin u_c=\operatorname{sech}(L/\ell)$ (\cref{thm:longthroat}); the BF analysis (\cref{prop:throatbf}); the entropy of the throat and the Cardy-type density of states (\cref{thm:attractor,prop:cardy}); the instanton action (\cref{thm:instanton}).\\
\textbf{It provides:} the reduction of spherically symmetric EYM to two-dimensional dilaton gravity with potential $U(\Phi)=2-2\ell^2/\Phi$; the throat as its constant-dilaton vacuum; the expansion around it, which is Jackiw--Teitelboim gravity with ground-state entropy $S_0=\pi\ell^2/G=4\pi^2/g^2$, exactly half the instanton action; the general dilaton in the throat and its three classes; the Unruh--Deser--Levin temperature; the Schwarzian thermodynamics with its exact density of states; and the status of the Yang--Mills mode in this two-dimensional description.
\end{usesbox}

Near-extremal black holes have a universal low-energy description: two-dimensional Jackiw--Teitelboim gravity on the $\mathrm{AdS}_2$ throat, whose boundary dynamics is governed by the Schwarzian action \cite{AlmheiriPolchinski2015,MaldacenaStanfordYang2016}. This chapter derives that description for our throat directly from the EYM action, with all coefficients fixed in terms of $G$ and $g$, and then uses it to describe the throat segment of the glued spacetime.

\section{Reduction to dilaton gravity}

For a metric $h_{ab}\dd x^a\dd x^b+R^2\dd\Omega^2$ in the diagonal gauge, write $\Phi=R^2$.

\begin{theorem}[Dilaton gravity from EYM]\label{thm:dilatongravity}\ver\
\begin{enumerate}[label=(\roman*)]
\item The four-dimensional scalar curvature satisfies $NAR^2\,{}^{(4)}\!R=NA\bigl(\Phi\,{}^{(2)}\!R+2+(\nabla\Phi)^2/(2\Phi)\bigr)-2\,\mathrm{Div}$, where $\mathrm{Div}=\partial_t\bigl(-AN^{-1}\partial_t\Phi\bigr)+\partial_x\bigl(NA^{-1}\partial_x\Phi\bigr)$ is a total derivative.
\item The magnetic Yang--Mills term is $F^a_{\mu\nu}F^{a\mu\nu}=2(1-w^2)^2/R^4$, which equals $2/\Phi^2$ for $w=0$.
\item Integrating over the sphere, the potential term is $\frac1{16\pi G}4\pi\cdot2-\frac1{4g^2}4\pi\Phi\frac2{\Phi^2}=\frac1{4G}U(\Phi)$ with
\[
U(\Phi)=2-\frac{2\ell^2}\Phi,\qquad \ell^2=\frac{4\pi G}{g^2},
\]
and $U'(\Phi)=2\ell^2/\Phi^2$. With $w$ restored, $-2\ell^2(1-w^2)^2/\ell^2=-2+4w^2-2w^4$ at $\Phi=\ell^2$.
\item On the throat the on-shell Lagrangian density has magnitude $F^2/(4g^2)=1/(2g^2\ell^4)=1/(8\pi G\ell^2)$; integrated over the sphere of area $4\pi\ell^2$ and a region of two-dimensional area $A$ it gives $\frac1{8\pi G\ell^2}\cdot4\pi\ell^2\cdot A=A/(2G)$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} Start from the warped-product formula of the proof of \cref{thm:ghy}, multiplied by $R^2$:
\[
R^2\,{}^{(4)}\!R=\Phi\,{}^{(2)}\!R-4R\,\Box R+2-2(\nabla R)^2 .
\]
Express everything through $\Phi=R^2$. From $\nabla\Phi=2R\nabla R$ we get $(\nabla R)^2=(\nabla\Phi)^2/(4\Phi)$. From $\Box\Phi=2R\Box R+2(\nabla R)^2$ we get $R\Box R=\frac12\Box\Phi-(\nabla\Phi)^2/(4\Phi)$. Substituting,
\[
R^2\,{}^{(4)}\!R=\Phi\,{}^{(2)}\!R+2+\frac{(\nabla\Phi)^2}{2\Phi}-2\Box\Phi .
\]
Multiply by $\sqrt{-h}=NA$. Then $NA\,\Box\Phi=\partial_\mu(NA\,h^{\mu\nu}\partial_\nu\Phi)$, and with $h^{tt}=-N^{-2}$, $h^{xx}=A^{-2}$ this is exactly $\mathrm{Div}$.

\emph{(ii)} By \cref{thm:ymstress}(i) the only $(\theta,\phi)$ component is $F^3_{\theta\phi}=(w^2-1)\sin\theta$. So $F^2\supset2g^{\theta\theta}g^{\phi\phi}(F^3_{\theta\phi})^2=2(1-w^2)^2/R^4$.

\emph{(iii)} Integrate $\sqrt{-g}=NAR^2\sin\theta$ over the sphere, which gives $4\pi\Phi\,NA$. The non-derivative terms per unit $NA$ are then
\[
\frac{4\pi\cdot2}{16\pi G}-\frac{4\pi\Phi}{4g^2}\cdot\frac2{\Phi^2}=\frac1{2G}-\frac{2\pi}{g^2\Phi}=\frac1{4G}\Bigl(2-\frac{8\pi G}{g^2\Phi}\Bigr),
\]
and $8\pi G/g^2=2\ell^2$. Differentiating, $U'=2\ell^2/\Phi^2$. With $w$ restored, the factor $1$ becomes $(1-w^2)^2=1-2w^2+w^4$.

\emph{(iv)} $F^2/(4g^2)=\frac2{4g^2\ell^4}$, and $\frac1{2g^2\ell^4}=\frac1{2g^2\ell^2}\cdot\frac{g^2}{4\pi G}=\frac1{8\pi G\ell^2}$. Multiplying by $4\pi\ell^2A$ gives $A/(2G)$.
\end{proof}


\begin{proposition}[The throat as a constant-dilaton vacuum]\label{prop:constdilaton}\ver\
The equations for constant $\Phi$ are $U(\Phi)=0$ and ${}^{(2)}\!R+U'(\Phi)=0$. Their unique solution is $\Phi=\ell^2$, ${}^{(2)}\!R=-2/\ell^2$, and it is realised: in synchronous coordinates $a=\cos(t/\ell)$ has $\ddot a=-a/\ell^2$, so ${}^{(2)}\!R=2\ddot a/a=-2/\ell^2$, an $\mathrm{AdS}_2$ of radius $\ell$. Moreover $U'(\ell^2)\ell^2=2$.
\end{proposition}
\begin{proof}
The two-dimensional Lagrangian is $\sqrt{-h}\bigl(\Phi\,{}^{(2)}\!R+U(\Phi)+(\nabla\Phi)^2/(2\Phi)\bigr)$.
\begin{itemize}
\item Varying $\Phi$ gives ${}^{(2)}\!R+U'(\Phi)$ plus terms containing $\nabla\Phi$, which vanish for constant $\Phi$.
\item Varying $h$, the term $\Phi\,{}^{(2)}\!R$ gives only $\nabla\nabla\Phi$ terms, since $\sqrt{-h}\,{}^{(2)}\!R$ is a total derivative in two dimensions. The term $\sqrt{-h}\,U$ gives $\frac12h_{ab}U$. For constant $\Phi$ only the latter survives, so $U=0$.
\end{itemize}
$U(\Phi)=2-2\ell^2/\Phi=0$ forces $\Phi=\ell^2$, and then ${}^{(2)}\!R=-U'(\ell^2)=-2/\ell^2$. For $a=\cos(t/\ell)$, $\ddot a=-a/\ell^2$, so $2\ddot a/a=-2/\ell^2$. Finally $U'(\ell^2)\ell^2=2\ell^2/\ell^4\cdot\ell^2=2$.
\end{proof}


\section{The Jackiw--Teitelboim expansion}

\begin{theorem}[JT gravity]\label{thm:jt}\ver\
\begin{enumerate}[label=(\roman*)]
\item Put $\Phi=\ell^2(1+\varphi)$. Then, exactly,
\[
\Phi\,{}^{(2)}\!R+U(\Phi)=\ell^2\,{}^{(2)}\!R+\varphi\bigl(\ell^2\,{}^{(2)}\!R+2\bigr)-\frac{2\varphi^2}{1+\varphi},
\]
and the kinetic term is $(\nabla\Phi)^2/(2\Phi)=\ell^2(\nabla\varphi)^2/(2(1+\varphi))$, of second order in $\varphi$.
\item To first order in $\varphi$ the action is the topological term $\ell^2\,{}^{(2)}\!R$ plus the JT term $\varphi(\ell^2\,{}^{(2)}\!R+2)$, whose equation for $\varphi$ fixes ${}^{(2)}\!R=-2/\ell^2$.
\item The ground-state entropy is
\[
S_0=\frac{4\pi\ell^2}{4G}=\frac{\pi\ell^2}G=\frac{4\pi^2}{g^2}=\frac12\cdot\frac{8\pi^2}{g^2}.
\]
With the two-dimensional Newton constant $G_2=G/(4\pi)$: $1/(16\pi G_2)=1/(4G)$, $S_0=\ell^2/(4G_2)$, and the Schwarzian coupling is $C=\varphi_r/(8\pi G_2)=\varphi_r/(2G)$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} $\Phi\,{}^{(2)}\!R=\ell^2\,{}^{(2)}\!R+\varphi\ell^2\,{}^{(2)}\!R$. For the potential,
\[
U=2-\frac2{1+\varphi}=\frac{2\varphi}{1+\varphi}=2\varphi-\frac{2\varphi^2}{1+\varphi},
\]
since $2\varphi(1+\varphi)-2\varphi^2=2\varphi$. Adding gives the stated form. For the kinetic term, $\nabla\Phi=\ell^2\nabla\varphi$, so $(\nabla\Phi)^2/(2\Phi)=\ell^4(\nabla\varphi)^2/(2\ell^2(1+\varphi))$.

\emph{(ii)} Drop the terms of second order. The coefficient of $\varphi$ is $\ell^2\,{}^{(2)}\!R+2$, and its vanishing is the equation for $\varphi$.

\emph{(iii)} $S_0=A/(4G)$ with $A=4\pi\ell^2$, and $\ell^2=4\pi G/g^2$ gives $\pi\ell^2/G=4\pi^2/g^2$. With $G_2=G/(4\pi)$: $16\pi G_2=4G$; $\ell^2/(4G_2)=\pi\ell^2/G$; and $\varphi_r/(8\pi G_2)=\varphi_r/(2G)$.
\end{proof}


The ground-state entropy of the throat is exactly half the action of a Yang--Mills instanton, $S_0=\frac12\,S_{\rm inst}$. Both are $1/g^2$ effects with geometric coefficients: $4\pi\ell^2$ is the area of the throat, $8\pi^2$ the volume of a unit $S^3$ times $4$.

\section{The dilaton in the throat}

\begin{theorem}[General dilaton]\label{thm:dilaton}\ver\
In synchronous coordinates $-\dd t^2+\cos^2(t/\ell)\dd z^2$ the dilaton equations ($tt$, $zz$, $tz$) are solved by
\[
\varphi=A\sin(t/\ell)+\cos(t/\ell)\bigl(B\cosh(z/\ell)+C\sinh(z/\ell)\bigr).
\]
\begin{enumerate}[label=(\roman*)]
\item (No static dilaton) If $\varphi_t=0$ everywhere then $A=B=C=0$.
\item (Invariant) $\varphi^2-\ell^2(\nabla\varphi)^2=A^2+B^2-C^2$. The vector $\xi^t=\varphi_z/\cos$, $\xi^z=-\varphi_t/\cos$ is a Killing vector with $\xi\cdot\xi=-(\nabla\varphi)^2$.
\item (Three classes) With $I=(A^2+B^2-C^2)/\ell^2$: the symmetric solution $A=C=0$ has $I=B^2/\ell^2>0$ (thermal); $A=0$, $B=C$ has $I=0$ (extremal); $A=B=0$, $C\ne0$ has $I<0$.
\item For the symmetric solution $\varphi=B\cos(t/\ell)\cosh(z/\ell)$, $(\nabla\varphi)^2=(B/\ell)^2(\sinh^2(z/\ell)-\sin^2(t/\ell)\cosh^2(z/\ell))$, which vanishes at the centre $(0,0)$: the bifurcation point of a Killing horizon.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{The solution.} Write $c=\cos(t/\ell)$, $s=\sin(t/\ell)$ and $Y=B\cosh(z/\ell)+C\sinh(z/\ell)$, so that $Y_{zz}=Y/\ell^2$ and $\varphi=As+cY$. With $a=c$, $a'/a=-s/(\ell c)$, and the metric equations of the JT action, $\nabla_a\nabla_b\varphi-h_{ab}\Box\varphi+h_{ab}\varphi/\ell^2=0$, read in these coordinates (with $\Gamma^t{}_{zz}=aa'$ and $\Gamma^z{}_{tz}=a'/a$):
\[
zz:\ \varphi_{tt}+\frac\varphi{\ell^2}=0,\qquad tz:\ \varphi_{tz}-\frac{a'}a\varphi_z=0,\qquad tt:\ -\frac{a'}a\varphi_t+\frac{\varphi_{zz}}{a^2}-\frac\varphi{\ell^2}=0 .
\]
Now $\varphi_t=(Ac-sY)/\ell$ and $\varphi_{tt}=-\varphi/\ell^2$, which is the $zz$ equation. Next, $\varphi_{tz}=-sY_z/\ell=(a'/a)\,cY_z$, which is the $tz$ equation. Multiplying the $tt$ equation by $\ell^2c$ gives
\[
Asc-s^2Y+Y-Asc-c^2Y=Y(1-s^2-c^2)=0 .
\]

\emph{(i)} At $t=0$, $\varphi_t=A/\ell$, so $A=0$. Then $\varphi_t=-sY/\ell=0$ for $t\ne0$ forces $Y\equiv0$, and so $B=C=0$.

\emph{(ii)} $(\nabla\varphi)^2=-\varphi_t^2+\varphi_z^2/c^2$. Then
\[
\varphi^2-\ell^2(\nabla\varphi)^2=(As+cY)^2+(Ac-sY)^2-\ell^2Y_z^2=A^2+Y^2-\ell^2Y_z^2 ,
\]
because the cross terms cancel and $s^2+c^2=1$. Also $Y^2-\ell^2Y_z^2=(B^2-C^2)(\cosh^2-\sinh^2)=B^2-C^2$.

For the Killing vector, the dilaton equation $\nabla_a\nabla_b\varphi-h_{ab}\Box\varphi+h_{ab}\varphi/\ell^2=0$ has trace $\Box\varphi=2\varphi/\ell^2$. So $\nabla_a\nabla_b\varphi=h_{ab}\varphi/\ell^2$. With $\xi_b=\epsilon_b{}^c\nabla_c\varphi$ we get $\nabla_a\xi_b=\epsilon_{ba}\varphi/\ell^2$, which is antisymmetric, so $\xi$ is Killing. In components $\xi=(\varphi_z/c,-\varphi_t/c)$, and $\xi\cdot\xi=-\varphi_z^2/c^2+\varphi_t^2=-(\nabla\varphi)^2$.

\emph{(iii)} Substitute into $(A^2+B^2-C^2)/\ell^2$.

\emph{(iv)} For $A=C=0$: $\varphi_t=-sB\cosh(z/\ell)/\ell$ and $\varphi_z=cB\sinh(z/\ell)/\ell$. Insert these into $(\nabla\varphi)^2$. At $(0,0)$ both terms vanish while $\varphi=B\ne0$.
\end{proof}


\begin{theorem}[Local temperature]\label{thm:deserlevin}\ver\
Let $N^2=\varphi^2/\ell^2-I$ be the norm of the Killing vector. Then the proper acceleration of its orbits is $a^2=\varphi^2/(\ell^2(\varphi^2-I\ell^2))$, and
\[
a^2-\frac1{\ell^2}=\frac{I}{\varphi^2-I\ell^2}.
\]
The surface gravity at the horizon $\varphi^2=I\ell^2$ is $\kappa^2=I/\ell^2$, and the local temperature squared $\kappa^2/(-\xi\cdot\xi)$ equals $a^2-1/\ell^2$ (times $1/4\pi^2$).
\end{theorem}
\begin{proof}
By \cref{thm:dilaton}(ii), $N^2=-\xi\cdot\xi=(\nabla\varphi)^2=\varphi^2/\ell^2-I$. The proper acceleration of the orbits of a Killing vector is $a=|\nabla N|/N$. Since $\nabla(N^2)=2\varphi\nabla\varphi/\ell^2$,
\[
|\nabla N|^2=\frac{\varphi^2(\nabla\varphi)^2}{\ell^4N^2}=\frac{\varphi^2}{\ell^4},\qquad a^2=\frac{\varphi^2}{\ell^4N^2}=\frac{\varphi^2}{\ell^2(\varphi^2-I\ell^2)}.
\]
Subtracting, $a^2-\frac1{\ell^2}=\frac{\varphi^2-(\varphi^2-I\ell^2)}{\ell^2(\varphi^2-I\ell^2)}=\frac I{\varphi^2-I\ell^2}$.

The surface gravity is $|\nabla N|$ on the horizon, where $\varphi^2=I\ell^2$; so $\kappa^2=I/\ell^2$. The local temperature is $T_{\rm loc}=\kappa/(2\pi N)$, and
\[
4\pi^2T_{\rm loc}^2=\frac{I/\ell^2}{\varphi^2/\ell^2-I}=\frac I{\varphi^2-I\ell^2}=a^2-\frac1{\ell^2}.\qedhere
\]
\end{proof}


This is the Deser--Levin form of the Unruh effect in anti-de Sitter space \cite{DeserLevin1997}: an observer in $\mathrm{AdS}_2$ detects a temperature only if $a>1/\ell$, and then $T_{\rm loc}=\sqrt{a^2-1/\ell^2}/(2\pi)$.

\begin{proposition}[Rindler observers and the junction]\label{prop:rindler}\ver\
\begin{enumerate}[label=(\roman*)]
\item The normalised Rindler lapse $N=\sinh(\rho/\ell)$ satisfies $N'=\cosh(\rho/\ell)/\ell$, $N''=N/\ell^2$ and $N'(0)=1/\ell$. The observer with $N=1$ sits at $\rho=\ell\log(1+\sqrt2)$, where $\cosh=\sqrt2$ and $a=\sqrt2/\ell$.
\item If the junction is at $\sin u_c=\operatorname{sech}(L/\ell)$ with $0<u_c<\pi/2$, then $\tan u_c=1/\sinh(L/\ell)$; and $u_c\le\pi e^{-L/\ell}$ (\cref{thm:longthroat}).
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{(i)} $(\sinh(\rho/\ell))'=\cosh(\rho/\ell)/\ell$, and differentiating again gives $N/\ell^2$; at $\rho=0$ the first derivative is $1/\ell$. $N=1$ means $\rho/\ell=\operatorname{arsinh}1=\log(1+\sqrt2)$, where $\cosh=\sqrt{1+1}$. Then $a=N'/N=\sqrt2/\ell$.

\emph{(ii)} For $0<u_c<\pi/2$, $\cos u_c=\sqrt{1-\operatorname{sech}^2}=\tanh(L/\ell)$. So $\tan u_c=\operatorname{sech}/\tanh=1/\sinh(L/\ell)$.
\end{proof}


\section{Schwarzian thermodynamics}

The boundary dynamics of JT gravity is the Schwarzian theory, whose one-loop exact partition function is $\log Z=S_0+\frac32\log C-\frac32\log\beta+2\pi^2C/\beta$ \cite{StanfordWitten2017} \ext.

\begin{theorem}[Thermodynamics and density of states]\label{thm:schwarzian}\ver\
\begin{enumerate}[label=(\roman*)]
\item $E=-\partial_\beta\log Z=3/(2\beta)+2\pi^2C/\beta^2=\frac32T+2\pi^2CT^2$, with heat capacity $\frac32+4\pi^2CT>0$.
\item $S=\log Z+\beta E=S_0+4\pi^2CT+\frac32\log T+\frac32(1+\log C)$.
\item For the exact density of states $\rho(E)\propto\sinh\bigl(2\pi\sqrt{2CE}\bigr)$, the inverse temperature is $1/T=4\pi^2C\coth x/x$ with $x=2\pi\sqrt{2CE}$, and
\[
\frac{4\pi^2C}x\le\frac{4\pi^2C\coth x}x\le\frac{4\pi^2C}x\Bigl(1+\frac2{e^{2x}-1}\Bigr),
\]
using $\coth x=1+2/(e^{2x}-1)$: the corrections to the leading Cardy-type behaviour are exponentially small.
\item The canonical energy $\frac32T+2\pi^2CT^2$ differs from the microcanonical leading term $2\pi^2CT^2$ by the relative amount $3/(4\pi^2CT)$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} $-\partial_\beta\bigl(-\frac32\log\beta+2\pi^2C/\beta\bigr)=\frac3{2\beta}+\frac{2\pi^2C}{\beta^2}$. Then $\dd E/\dd T=\frac32+4\pi^2CT$.

\emph{(ii)} $\log Z+\beta E=S_0+\frac32\log C+\frac32\log T+2\pi^2CT+\frac32+2\pi^2CT$.

\emph{(iii)} $S(E)=\log\rho$ up to a constant. Since $\sqrt{2C/E}=4\pi C/x$,
\[
\frac{\dd x}{\dd E}=\frac{2\pi\sqrt{2C}}{2\sqrt E}=\frac{4\pi^2C}x .
\]
So $1/T=\coth x\cdot\dd x/\dd E=4\pi^2C\coth x/x$. Finally, $\coth x=\frac{e^{2x}+1}{e^{2x}-1}=1+\frac2{e^{2x}-1}\ge1$.

\emph{(iv)} $\frac{3T/2}{2\pi^2CT^2}=\frac3{4\pi^2CT}$.
\end{proof}


\section{The Yang--Mills mode and the symmetry}

\begin{proposition}\label{prop:jtmode}\ver\
\begin{enumerate}[label=(\roman*)]
\item $\Delta(\Delta-1)=-1$ has the roots $\Delta=\frac12\pm i\frac{\sqrt3}2$ and no real root; the Yang--Mills mode has $m^2\ell^2=-1$, below the BF bound.
\item For $a\ge0$, $e^{ax^2}$ is not integrable on $\mathbb R$: the Gaussian integral along a negative mode of the Euclidean action diverges.
\item An $SL(2,\mathbb R)$-invariant configuration has $r=\Omega\sin\chi$ constant; if it vanishes at a pole, it vanishes identically. A regular closed universe cannot be $SL(2,\mathbb R)$-invariant.
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{(i)} $\Delta^2-\Delta+1=0$ has discriminant $1-4=-3$, so its roots are $\frac12\pm\frac{\sqrt3}2\ii$.

\emph{(ii)} For $a\ge0$, $e^{ax^2}\ge1$, so $\int_{-L}^Le^{ax^2}\ge2L\to\infty$.

\emph{(iii)} $SL(2,\mathbb R)$ acts transitively on $\mathrm{AdS}_2$, and an invariant configuration has $r$ constant on each orbit, hence constant on the whole two-dimensional part. If $r=\Omega\sin\chi=0$ at a pole, this constant is $0$.
\end{proof}


The JT description therefore applies to the gravitational sector of the throat, but the Yang--Mills mode is not part of a stable two-dimensional theory: its Euclidean path integral has a divergent direction, which is the two-dimensional shadow of the instability of \cref{ch:stability}. And the glued spacetime as a whole, which closes off at the poles of the Tolman regions, necessarily breaks the $SL(2,\mathbb R)$ symmetry of the throat.
