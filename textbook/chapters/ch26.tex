\chapter{Canonical Quantisation and the Wheeler--DeWitt Equation}\markboth{\MakeUppercase{Chapter 26.\ Canonical quantisation}}{}
\label{ch:wdw}

\begin{usesbox}
\textbf{This chapter uses:} the reduced action and the Goursat form (\cref{thm:redEL,thm:goursat}); the throat in synchronous coordinates, which is a Kantowski--Sachs cosmology (\cref{thm:throatevo}); the fronts $z=\ell\operatorname{arsinh}\tan(t/\ell)$ (\cref{thm:fronts}); the minisuperspace energy balance on $S^3$ (\cref{prop:minisuper}); the instanton (\cref{thm:instanton}).\\
\textbf{It provides:} the canonical form of spherically symmetric EYM, with Kucha\v{r}'s mass variable and the observation that it degenerates exactly on the throat; a single constraint system that contains both the Friedmann constraint of the Tolman side and the Kantowski--Sachs constraint of the throat; exact solutions of the Wheeler--DeWitt equation of the throat, for two operator orderings; a Dirac observable $K$ and a clock $T$ with $\{T,K\}=1$; the reduced phase space as a half-plane and its affine quantisation; the semiclassical emergence of time from stationary phase; and an honest discussion of a semiclassical level condition for the Yang--Mills coupling.
\end{usesbox}

The classical analysis of Parts~IV and~V ended with a glued spacetime that collapses. The natural next question is what quantum mechanics does to it. The full quantum theory of gravity is not available, but in spherical symmetry the constraints reduce to a field theory in two dimensions, and on the throat to a minisuperspace with two degrees of freedom. Both can be quantised in the Dirac way, as Wheeler and DeWitt proposed \cite{DeWitt1967}. This chapter does so, proving exact statements where the formalism allows it and labelling the rest as external input.

\section{The reduced action and its canonical form}

In units $G=1$ the action reduced over the sphere is $\frac14\int\dd^2x\sqrt{-h}\,\{r^2R_h+2(\nabla r)^2+2-4\ell^2(\nabla w)^2-2\ell^2(1-w^2)^2/r^2\}$.

\begin{proposition}[Euler--Lagrange equations]\label{prop:canEL}\ver\
In conformal gauge $h=-2\Omega\,\dd u\,\dd v$, the Euler--Lagrange equations for $\log\Omega$, $w$ and $r$ are
\begin{align*}
&2(r^2)_{uv}+2\Omega-2\ell^2\Omega(1-w^2)^2/r^2=0,\qquad -16\ell^2w_{uv}+8\ell^2\Omega w(1-w^2)/r^2=0,\\
&4r(\log\Omega)_{uv}+8r_{uv}+4\ell^2\Omega(1-w^2)^2/r^3=0,
\end{align*}
and all three hold whenever the Goursat equations of \cref{thm:goursat} hold. (The variational calculus itself is standard \ext.)
\end{proposition}
\begin{proof}
\emph{The Lagrangian in conformal gauge.} For $h=-2\Omega\,\dd u\,\dd v$ we have $\sqrt{-h}=\Omega$ and $h^{uv}=-1/\Omega$. By the proof of \cref{thm:goursat}, $\sqrt{-h}R_h=2(\log\Omega)_{uv}$, and $\sqrt{-h}(\nabla f)^2=-2f_uf_v$. With $\lambda=\log\Omega$ and $P=(1-w^2)^2$ the Lagrangian is
\[
\tfrac14\bigl\{2r^2\lambda_{uv}-4r_ur_v+2e^\lambda+8\ell^2w_uw_v-2\ell^2e^\lambda P/r^2\bigr\}.
\]

\emph{The three Euler--Lagrange equations}, each multiplied by $4$:
\begin{itemize}[nosep]
\item $\lambda$ (the term $r^2\lambda_{uv}$ is differentiated twice): $2(r^2)_{uv}+2\Omega-2\ell^2\Omega P/r^2$.
\item $w$: $\partial L/\partial w=2\ell^2\Omega w(1-w^2)/r^2$ and $\partial_u(2\ell^2w_v)+\partial_v(2\ell^2w_u)=4\ell^2w_{uv}$, which gives $8\ell^2\Omega w(1-w^2)/r^2-16\ell^2w_{uv}$.
\item $r$: $\partial L/\partial r=r\lambda_{uv}+\ell^2\Omega P/r^3$, and the $r_ur_v$ term contributes $+2r_{uv}$, which gives $4r\lambda_{uv}+8r_{uv}+4\ell^2\Omega P/r^3$.
\end{itemize}

\emph{Consistency with the Goursat equations.} The first follows from $(r^2)_{uv}=2(rr_{uv}+r_ur_v)$ and the mixed equation. The second is \cref{thm:ymnull} multiplied by $-16\ell^2$. The third follows from the $\theta\theta$ equation divided by $\Omega$, which reads $\lambda_{uv}=-2r_{uv}/r-\Omega\ell^2P/r^4$.
\end{proof}


With $h=-N^2\dd t^2+\Lambda^2(\dd x+N^x\dd t)^2$ the Hamiltonian and momentum constraints are
\[
H=-\frac{P_rP_\Lambda}r+\frac{\Lambda P_\Lambda^2}{2r^2}+\frac{rr''}\Lambda-\frac{rr'\Lambda'}{\Lambda^2}+\frac{r'^2}{2\Lambda}-\frac\Lambda2+H_m,\qquad H_x=P_rr'-\Lambda P_\Lambda'+P_ww'.
\]

\begin{theorem}[Kucha\v{r}'s mass]\label{thm:kuchar}\ver\
Let $M=P_\Lambda^2/(2r)+\frac r2(1-r'^2/\Lambda^2)$.
\begin{enumerate}[label=(\roman*)]
\item $M'=-(r'/\Lambda)H_g-(P_\Lambda/(r\Lambda))H_{x,g}$, so on the constraint surface $M'=(r'/\Lambda)H_m+(P_\Lambda/(r\Lambda))P_ww'$; without matter, $M'=0$ (Birkhoff's theorem in canonical form).
\item $F:=r'^2/\Lambda^2-P_\Lambda^2/r^2=1-2M/r$.
\item With $P_M=\Lambda P_\Lambda/(rF)$ one has $\Lambda^2=r'^2/F-FP_M^2$ and $P_\Lambda/(r\Lambda)=FP_M/\Lambda^2$ wherever $F\ne0$.
\item On the throat, $r'=0$ and $P_\Lambda=0$, so $F=0$: Kucha\v{r}'s variables \cite{Kuchar1994} are not available exactly there.
\end{enumerate}
\end{theorem}
\begin{proof}
Write $P=P_\Lambda$, and let $H_g$, $H_{x,g}$ be the gravitational parts of the constraints.

\emph{(i)} Differentiate $M$:
\[
M'=-\frac{P^2r'}{2r^2}+\frac{PP'}r+\frac{r'}2\Bigl(1-\frac{r'^2}{\Lambda^2}\Bigr)-\frac{rr'r''}{\Lambda^2}+\frac{rr'^2\Lambda'}{\Lambda^3}.
\]
On the other hand,
\begin{align*}
-\frac{r'}\Lambda H_g&=\frac{P_rPr'}{r\Lambda}-\frac{r'P^2}{2r^2}-\frac{rr'r''}{\Lambda^2}+\frac{rr'^2\Lambda'}{\Lambda^3}-\frac{r'^3}{2\Lambda^2}+\frac{r'}2,\\
-\frac P{r\Lambda}H_{x,g}&=-\frac{PP_rr'}{r\Lambda}+\frac{PP'}r .
\end{align*}
The $P_r$ terms cancel, and the rest matches $M'$ term by term. On the constraint surface $H_g=-H_m$ and $H_{x,g}=-P_ww'$, which gives the second form.

\emph{(ii)} $1-\frac{2M}r=1-\frac{P^2}{r^2}-1+\frac{r'^2}{\Lambda^2}$.

\emph{(iii)} With $P_M=\Lambda P/(rF)$,
\[
\frac{r'^2}F-FP_M^2=\frac{\Lambda^2}F\Bigl(\frac{r'^2}{\Lambda^2}-\frac{P^2}{r^2}\Bigr)=\Lambda^2 ,
\]
and $FP_M/\Lambda^2=P/(r\Lambda)$.

\emph{(iv)} $F=0-0$.
\end{proof}


\begin{proposition}[One constraint system for both pieces]\label{prop:bothconstraints}\ver\
\begin{enumerate}[label=(\roman*)]
\item Inserting the Tolman form $\Lambda=\Omega$, $r=\Omega\sin\chi$, $w=\cos\chi$, $P_w=0$, $P_\Lambda=-\Omega_\eta\sin^2\chi$, $P_r=-2\Omega_\eta\sin\chi$ gives $H=-\frac{3\sin^2\chi}{2\Omega}(\Omega_\eta^2+\Omega^2-\ell^2)$ and $H_x=0$: the Friedmann constraint.
\item For fields independent of $x$, $H$ is twice the Kantowski--Sachs constraint of the throat side (below).
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{(i)} Here $r'=\Omega\cos\chi$, $r''=-\Omega\sin\chi$ and $\Lambda'=0$. Term by term:
\begin{itemize}[nosep]
\item $-P_rP_\Lambda/r=-2\Omega_\eta^2\sin^2\chi/\Omega$ and $\Lambda P_\Lambda^2/(2r^2)=\Omega_\eta^2\sin^2\chi/(2\Omega)$, together $-\frac32\Omega_\eta^2\sin^2\chi/\Omega$;
\item $rr''/\Lambda+r'^2/(2\Lambda)-\Lambda/2=-\Omega\sin^2\chi+\frac\Omega2(\cos^2\chi-1)=-\frac32\Omega\sin^2\chi$;
\item with $P_w=0$ the Yang--Mills part is $H_m=\ell^2w'^2/\Lambda+\ell^2\Lambda(1-w^2)^2/(2r^2)=\frac32\ell^2\sin^2\chi/\Omega$.
\end{itemize}
The sum is the stated $H$. For the momentum constraint, $H_x=-2\Omega_\eta\Omega\sin\chi\cos\chi+\Omega\cdot2\Omega_\eta\sin\chi\cos\chi+0=0$.

\emph{(ii)} For $x$-independent fields, $H=-P_rP_\Lambda/r+\Lambda P_\Lambda^2/(2r^2)-\Lambda/2+P_w^2/(4\ell^2\Lambda)+\ell^2\Lambda(1-w^2)^2/(2r^2)$. Compare with the Kantowski--Sachs constraint of \cref{thm:ks}, with $c=\frac12$ and potential $-c^2a(1-\ell^2(1-w^2)^2/r^2)$. Every term is exactly half the corresponding term of $H$.
\end{proof}


\section{The throat as a Kantowski--Sachs minisuperspace}

In synchronous coordinates the throat is homogeneous along $z$: $-N^2\dd t^2+a(t)^2\dd x^2+r(t)^2\dd\Omega^2$, with $w=w(t)$. With $c=L_x/(2G)$ for a coordinate length $L_x$, the reduced action is $c\int\dd t\{[-a\dot r^2-2r\dot a\dot r+2\ell^2a\dot w^2]/N+Na[1-\ell^2(1-w^2)^2/r^2]\}$.

\begin{theorem}[Kantowski--Sachs Hamiltonian]\label{thm:ks}\ver\
\begin{enumerate}[label=(\roman*)]
\item With $p_a=-2cr\dot r$, $p_r=-2c(a\dot r+r\dot a)$, $p_w=4c\ell^2a\dot w$, the kinetic Hamiltonian $-p_ap_r/(2r)+ap_a^2/(4r^2)+p_w^2/(8\ell^2a)$ equals $c^2(-a\dot r^2-2r\dot a\dot r+2\ell^2a\dot w^2)$. It is linear in $p_r$: $r$ can serve as a clock.
\item The RN interior $f=1-2M/r+\ell^2/r^2$ satisfies the constraint in the form $f+rf'=1-\ell^2/r^2$.
\item For $w=0$, with $\sigma=a^2r/2$ and $\theta=-(r+\ell^2/r)/2$, the function $S=k\sigma+\kappa\theta$ solves the Hamilton--Jacobi equation when $k\kappa=4c^2$. The Wheeler--DeWitt equation takes the form $\partial_\sigma\partial_\theta\Psi=-4c^2\Psi$, solved by $\cos(k\sigma+\kappa\theta)$; the plane waves $e^{i(k\sigma+\kappa\theta)}$ have unit modulus.
\item (Inversion) $r\mapsto\ell^2/r$, $a\mapsto ar/\ell$ leaves $\sigma$ and $\theta$ invariant.
\item (Fold) $r+\ell^2/r-2\ell=(r-\ell)^2/r$, so $\theta\le-\ell$ with equality iff $r=\ell$: the throat is the fold of the minisuperspace under the inversion.
\item The potential $\ell^2(1-w^2)^2=\ell^2-2\ell^2w^2+\ell^2w^4$ is an inverted oscillator in $w$ at $w=0$.
\item The proper length $L=aL_x$ of the throat and $p_L=p_a/L_x=-r\dot r$ are canonically conjugate, $p_LL=p_aa$; consequently $\Delta L\,\Delta(r\dot r)\ge\hbar/2$ \ext.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} Insert the momenta:
\begin{align*}
-\frac{p_ap_r}{2r}&=-2c^2a\dot r^2-2c^2r\dot a\dot r,\\
\frac{ap_a^2}{4r^2}&=c^2a\dot r^2,\\
\frac{p_w^2}{8\ell^2a}&=2c^2\ell^2a\dot w^2 .
\end{align*}
The sum is $c^2(-a\dot r^2-2r\dot a\dot r+2\ell^2a\dot w^2)$. Only the first term contains $p_r$, and it does so linearly.

\emph{(ii)} $rf'=2M/r-2\ell^2/r^2$, so in $f+rf'$ the $M$ terms cancel and $1-\ell^2/r^2$ remains.

\emph{(iii)} With $S=k\sigma+\kappa\theta$: $p_a=\partial_aS=kar$ and $p_r=\partial_rS=\frac k2a^2-\frac\kappa2(1-\ell^2/r^2)$. Then
\[
-\frac{p_ap_r}{2r}+\frac{ap_a^2}{4r^2}=-\frac{k^2a^3}4+\frac{k\kappa a}4\Bigl(1-\frac{\ell^2}{r^2}\Bigr)+\frac{k^2a^3}4 .
\]
The constraint $\frac{k\kappa}4a(1-\ell^2/r^2)-c^2a(1-\ell^2/r^2)=0$ holds exactly when $k\kappa=4c^2$. For the plane waves, $\partial_\sigma\partial_\theta\cos(k\sigma+\kappa\theta)=-k\kappa\cos(\cdot)$, and $|e^{\ii x}|=1$ for real $x$.

\emph{(iv)} $\sigma\mapsto\frac12\frac{a^2r^2}{\ell^2}\cdot\frac{\ell^2}r=\sigma$, and $\theta\mapsto-\frac12(\ell^2/r+r)=\theta$.

\emph{(v)} $(r-\ell)^2/r=(r^2-2r\ell+\ell^2)/r$. This is $\ge0$ for $r>0$, with equality only at $r=\ell$.

\emph{(vi)} Expand the square. In the potential energy $+\ell^2a(1-w^2)^2/r^2$ the coefficient of $w^2$ is $-2\ell^2a/r^2<0$.

\emph{(vii)} With $L=aL_x$, $\dot a=\dot L/L_x$, so $p_L=\partial L/\partial\dot L=p_a/L_x$, and $p_LL=p_aa$. With $c=L_x/2$ (in units $G=1$), $p_L=-r\dot r$. The uncertainty relation is the Robertson inequality for $[\hat L,\hat p_L]=\ii\hbar$ \ext.
\end{proof}


Part (vii) makes precise the remark of \cref{ch:envelope}: the free length $L$ of the glued slice is conjugate to the rate $-r\dot r=-\frac12\,\dd(r^2)/\dd t$ at which the throat's area changes.

\section{Exact solutions of the Wheeler--DeWitt equation}

In variables $(a,b)$ with $b=r$ the Wheeler--DeWitt operator of the throat is
\[
\hat H\Psi=\frac1{4c}\bigl(-\hbar^2a\,\Psi_{aa}+2\hbar^2b\,\Psi_{ab}\bigr)-c\,a(b^2-\ell^2)\Psi .
\]
Let $Q_C(b)=Cb-4c^2(b^2+\ell^2)$.

\begin{theorem}[Exact WKB solutions]\label{thm:wdwexact}\ver\
\begin{enumerate}[label=(\roman*)]
\item For $\Psi=A(b)e^{ia\sigma(b)/\hbar}$, the WKB expansion terminates: $\hat H\Psi=e^{ia\sigma/\hbar}\bigl(A\cdot\mathrm{HJ}+\frac{i\hbar b}{2c}(\sigma A'+\sigma'A)\bigr)$, where $\mathrm{HJ}$ is the Hamilton--Jacobi expression. All derivatives are the actual partial derivatives of the function.
\item At every $b$ with $Q_C(b)>0$, $\Psi_C=e^{ia\sigma/\hbar}/\sigma$ with $\sigma=\sqrt{Q_C}$ is an exact solution; so is the branch $\sigma=-\sqrt{Q_C}$. More generally the conditions are $\sigma^2=Q_C$, $\sigma\sigma'=Q_C'/2$, $A=1/\sigma$.
\item In the Laplace--Beltrami ordering, $\Psi=\sqrt b\,e^{ia\sigma/\hbar}/\sigma$ ($b>0$) is an exact solution for both signs of $\sigma$; the transport equation becomes $(\sigma A)'=(\sqrt b)'$.
\item The operator is linear in $(\Psi,\Psi_{aa},\Psi_{ab})$, so superpositions of solutions are solutions; $|\Psi|=|A|$, which diverges at the caustic $Q_C\to0$.
\item At the level of the general WKB expansion, $D^2(A\psi_0)=\psi_0[(ik)^2A(DS)^2+ik(2DA\,DS+AD^2S)+D^2A]$ for $D\psi_0=ik\,DS\,\psi_0$; in two dimensions the $(ik)^2$ coefficient of $\Box(A\psi_0)$ is $A(-S_t^2+wS_z^2)$, the eikonal.
\item The throat front $S=z-\ell\operatorname{arsinh}\tan(t/\ell)$ has $S_t=-1/\cos(t/\ell)$, $S_z=1$ and satisfies the eikonal equation $-S_t^2+S_z^2/\cos^2(t/\ell)=0$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} Write $e=e^{\ii a\sigma/\hbar}$. Then $\Psi_a=\frac{\ii\sigma}\hbar\Psi$ and $\Psi_{aa}=-\frac{\sigma^2}{\hbar^2}\Psi$. Differentiating $\Psi_b=(A'+\ii a\sigma'A/\hbar)e$ in $a$,
\[
\Psi_{ab}=e\Bigl(\frac{\ii\sigma'A}\hbar+\frac{\ii\sigma A'}\hbar-\frac{a\sigma\sigma'A}{\hbar^2}\Bigr).
\]
Insert these into $\hat H$. The terms without $\hbar$ form $A\cdot\mathrm{HJ}$ with
\[
\mathrm{HJ}=\frac a{4c}(\sigma^2-2b\sigma\sigma')-ca(b^2-\ell^2),
\]
and the terms with one $\hbar$ are $\frac{\ii\hbar b}{2c}(\sigma'A+\sigma A')$. There are no further terms.

\emph{(ii)} If $\sigma^2=Q_C$, then $2\sigma\sigma'=Q_C'$, and
\[
Q_C-bQ_C'=Cb-4c^2b^2-4c^2\ell^2-Cb+8c^2b^2=4c^2(b^2-\ell^2).
\]
So $\mathrm{HJ}=\frac a{4c}\cdot4c^2(b^2-\ell^2)-ca(b^2-\ell^2)=0$. With $A=1/\sigma$ the transport term $(\sigma A)'$ vanishes. Both signs of $\sigma$ work.

\emph{(iii)} The Laplace--Beltrami operator (\cref{thm:kslegendre}(v)) has the extra term $-\frac{\hbar^2}{4c}\Psi_a=-\frac{\ii\hbar\sigma A}{4c}e$. The transport condition becomes $b(\sigma A)'=\frac12\sigma A$. With $\sigma A=\sqrt b$ both sides equal $\frac12\sqrt b$.

\emph{(iv)} $\hat H$ is linear. $|e|=1$ for real $\sigma$, so $|\Psi|=|A|=1/|\sigma|$, which diverges as $Q_C\to0$.

\emph{(v)} Apply the product rule twice, using $D\psi_0=\ii k\,DS\,\psi_0$. In two dimensions with $\Box=-\partial_t^2+w\,\partial_z^2$ the $(\ii k)^2$ terms are $A(-S_t^2+wS_z^2)$.

\emph{(vi)} By the derivative computed in \cref{thm:fronts}(i), $S_t=-\ell\cdot\frac1\ell\sec(t/\ell)$, and $S_z=1$. Then $-\sec^2+\sec^2=0$.
\end{proof}


Part (vi) connects the characteristic fronts of \cref{thm:fronts} with the eikonal of the wave equation: the Mercator lines are the rays of the semiclassical wave functions.

\section{Observables and the reduced phase space}

\begin{theorem}[A clock and an energy]\label{thm:observables}\ver\
Let $K=(p_a^2+4c^2(b^2+\ell^2))/b$ and $T=ab/(2p_a)$.
\begin{enumerate}[label=(\roman*)]
\item $\{T,K\}=1$ and $\{K,H\}=0$: $K$ is a Dirac observable. $\{T,H\}=\frac{b}{2p_a^2}H$, so $T$ is an observable on the constraint surface.
\item $\hat K=(-\hbar^2\partial_a^2+4c^2(b^2+\ell^2))/b$ satisfies $\hat K\Psi_C=C\Psi_C$.
\item $K-8c^2\ell=(p_a^2+4c^2(b-\ell)^2)/b$, so for $b>0$, $p_a\ne0$: $K>K_0=8c^2\ell$. The minimum $K_0$ is attained at $b=\ell$, $p_a=0$: the static throat.
\item $Q_C(b)>0$ for some $b$ iff $C^2>64c^4\ell^2$, and for some $b>0$ iff $C>8c^2\ell=K_0$ (for $c,\ell>0$).
\item For $a,b>0$ the sign of $T$ is the sign of $p_a$: the two sectors are expanding and contracting throats.
\end{enumerate}
\end{theorem}
\begin{proof}
The classical constraint read off from $\hat H$ is
\[
H=\frac a{4c}p_a^2-\frac b{2c}p_ap_b-ca(b^2-\ell^2).
\]

\emph{(i)} $T$ and $K$ do not involve $p_b$, and $K$ does not involve $a$. So
\[
\{T,K\}=\partial_aT\,\partial_{p_a}K=\frac b{2p_a}\cdot\frac{2p_a}b=1 .
\]
For $\{K,H\}$, with $K=p_a^2/b+4c^2b+4c^2\ell^2/b$:
\[
\{K,H\}=-\partial_{p_a}K\,\partial_aH+\partial_bK\,\partial_{p_b}H=-\frac{2p_a}b\Bigl(\frac{p_a^2}{4c}-c(b^2-\ell^2)\Bigr)-\frac{bp_a}{2c}\Bigl(-\frac{p_a^2}{b^2}+4c^2-\frac{4c^2\ell^2}{b^2}\Bigr).
\]
The two brackets cancel term by term. For $\{T,H\}$, the same bookkeeping gives
\[
\frac{ab}{8c}-\frac{b^2p_b}{4cp_a}-\frac{abc(b^2-\ell^2)}{2p_a^2},
\]
which is $\frac b{2p_a^2}H$.

\emph{(ii)} $-\hbar^2\partial_a^2$ acting on $\Psi_C$ gives $\sigma^2\Psi_C=Q_C\Psi_C$, and $Q_C+4c^2(b^2+\ell^2)=Cb$.

\emph{(iii)} $K-8c^2\ell=(p_a^2+4c^2(b^2-2b\ell+\ell^2))/b$. This is $\ge0$ for $b>0$, and $>0$ unless $p_a=0$ and $b=\ell$.

\emph{(iv)} $Q_C$ is a downward parabola in $b$ with vertex at $b=C/(8c^2)$ and maximum $C^2/(16c^2)-4c^2\ell^2$. This maximum is positive exactly when $C^2>64c^4\ell^2$. The positive point can be chosen with $b>0$ exactly when the vertex is at $b>0$, that is, $C>0$, so $C>8c^2\ell$.

\emph{(v)} $a,b>0$, so $T=ab/(2p_a)$ has the sign of $p_a$.
\end{proof}


\section{Affine quantisation}

The reduced phase space $\{K>K_0,\ T\in\mathbb R\}$ is a half-plane. Canonical quantisation of $(T,K)$ would allow $K<K_0$. The affine variables $K-K_0$ and $D=(K-K_0)T$ respect the half-plane: $\{D,K\}=K-K_0$, the Lie algebra of the affine group \cite{Klauder2012}.

\begin{theorem}[Affine representation]\label{thm:affine}\ver\
On $L^2((0,\infty),\dd k)$ let $\hat K=K_0+k$ and $\hat D=i\hbar(k\partial_k+\frac12)$.
\begin{enumerate}[label=(\roman*)]
\item $[\hat D,\hat K]=i\hbar(\hat K-K_0)$ pointwise, and $\langle\psi,(\hat K-K_0)\psi\rangle=\int_0^\infty k|\psi|^2\dd k\ge0$.
\item The operators $U(\alpha,\tau)$, combining the dilation $\psi(k)\mapsto\sqrt\alpha\,\psi(\alpha k)$ ($\alpha>0$) and the phase $e^{i\tau k/\hbar}$, satisfy $U(\alpha,\tau)U(\beta,\sigma)=U(\alpha\beta,\tau+\alpha\sigma)$ and preserve the norm.
\item The physical inner product is positive definite: $\langle f,f\rangle\ge0$ with equality iff $f=0$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} For a test function $\psi$,
\[
[\hat D,\hat K]\psi=\ii\hbar\bigl((k\partial_k+\tfrac12)(k\psi)-k(k\partial_k+\tfrac12)\psi\bigr)=\ii\hbar\,k\psi=\ii\hbar(\hat K-K_0)\psi .
\]
Also $\int k|\psi|^2\ge0$ because $k>0$.

\emph{(ii)} Define $(U(\alpha,\tau)\psi)(k)=e^{\ii\tau k/\hbar}\sqrt\alpha\,\psi(\alpha k)$. Then
\[
U(\alpha,\tau)U(\beta,\sigma)\psi(k)=e^{\ii\tau k/\hbar}\sqrt\alpha\,e^{\ii\sigma\alpha k/\hbar}\sqrt\beta\,\psi(\beta\alpha k)=U(\alpha\beta,\tau+\alpha\sigma)\psi(k).
\]
The substitution $k'=\alpha k$ shows $\int\alpha|\psi(\alpha k)|^2\dd k=\int|\psi|^2$.

\emph{(iii)} $\langle f,f\rangle=\int|f|^2\ge0$, and it vanishes only for $f=0$ in $L^2$.
\end{proof}


\section{Time from stationary phase}

Superpose the exact solutions, $\Psi=\int\dd C\,\phi(C)\Psi_C$. The phase $S_C=a\sqrt{Q_C(b)}$ is stationary when $\partial_CS_C=T_0$.

\begin{theorem}[Semiclassical clock]\label{thm:semiclock}\ver\
\begin{enumerate}[label=(\roman*)]
\item $\partial_CQ_C=b$ and $\partial_C(a\sqrt{Q_C})=ab/(2\sqrt{Q_C})$, which is $T$ evaluated at $p_a=\sigma$.
\item For $b>0$, $Q_C(b)>0$ iff $C>C_{\min}(b)=4c^2(b^2+\ell^2)/b$, and $C_{\min}(b)\ge8c^2\ell=K_0$; the domain in $C$ is the half-line $(C_{\min}(b),\infty)$. If the support of $\phi$ lies above $C_1>C_{\min}(b)$, all components are defined.
\item For $a,b>0$, $T(C)=ab/(2\sqrt{Q_C})$ is strictly decreasing, so the stationary point is unique; its second derivative $-ab^2/(4\sigma^3)$ is non-zero.
\item For every $T_0>0$ the stationary point exists: $C_*=C_{\min}(b)+a^2b/(4T_0^2)$ gives $Q=(ab/(2T_0))^2$ and $\partial_C(a\sqrt Q)=T_0$.
\end{enumerate}
\end{theorem}
\begin{proof}
\emph{(i)} $Q_C$ is linear in $C$ with coefficient $b$, and the chain rule gives $\partial_C\sqrt{Q_C}=b/(2\sqrt{Q_C})$.

\emph{(ii)} $Q_C>0$ means $Cb>4c^2(b^2+\ell^2)$, that is, $C>C_{\min}$ when $b>0$. By the inequality between arithmetic and geometric means, $(b^2+\ell^2)/b\ge2\ell$.

\emph{(iii)} $Q_C$ increases in $C$, so $T(C)=ab/(2\sqrt{Q_C})$ decreases. Explicitly, $\partial_CT=-ab\cdot b/(4Q_C^{3/2})=-ab^2/(4\sigma^3)\ne0$.

\emph{(iv)} $Q_{C_*}=(C_*-C_{\min})b=a^2b^2/(4T_0^2)$. So $\sqrt{Q}=ab/(2T_0)$, and by (i) $\partial_C(a\sqrt Q)=T_0$.
\end{proof}


Time is not an external parameter here: it is the variable conjugate to the Dirac observable $K$, and a wave packet in $K$ is peaked, by stationary phase, on the classical trajectory with clock reading $T_0$.

\section{The homogeneous sector on \texorpdfstring{$S^3$}{S3} and a level condition}

\begin{proposition}[Classical facts]\label{prop:s3sector}\ver\
\begin{enumerate}[label=(\roman*)]
\item The $S^3$ phase is not ultrastatic without a cosmological constant: $R_{tt}=0$ is incompatible with $R_{tt}=8\pi G\rho>0$ for traceless Yang--Mills stress.
\item For the reduced system $a''=-a$, $w''=2w(1-w^2)$, the constraint $a'^2+a^2-\ell^2(w'^2+(1-w^2)^2)$ is conserved.
\item The homogeneous Yang--Mills potential $V=K(1-f^2)^2$ has $V=K$ at the top $f=0$ and, with mass $2K$, oscillation frequency $2$ at the bottom $f=1$, where $V''=8K$. The instanton action is $3\pi^2\cdot\frac83=8\pi^2$.
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{(i)} For traceless $T$, $R_{tt}=8\pi G(T_{tt}-\frac12Tg_{tt})=8\pi G\rho>0$. For an ultrastatic metric $-\dd t^2+g_{S^3}$, $R_{tt}=0$.

\emph{(ii)} Differentiate the constraint:
\[
2a'(a''+a)-2\ell^2w'\bigl(w''-2w(1-w^2)\bigr)=0
\]
by the two equations of motion.

\emph{(iii)} $V(0)=K$. Next, $V'=-4Kf(1-f^2)$ and $V''=-4K(1-f^2)+8Kf^2$, which is $8K$ at $f=1$. The frequency is $\sqrt{V''/m}=\sqrt{8K/(2K)}=2$. Finally $3\pi^2\cdot\frac83=8\pi^2$.
\end{proof}


The gravitational part of the constraint is a harmonic oscillator of negative energy (\cref{prop:minisuper}), $H_\Omega=-\frac{3\pi}{4G}(\dot\Omega^2+\Omega^2)$.

\begin{proposition}[Oscillator levels]\label{prop:levels}\ver\
\begin{enumerate}[label=(\roman*)]
\item $\psi=e^{-M\Omega^2/2}$ satisfies $-\frac1{2M}\psi''+\frac M2\Omega^2\psi=\frac12\psi$ ($\hbar=1$, frequency $1$); for $H_\Omega$ the levels are $-(n+\frac12)$ \ext.
\item Setting the classical energy $\frac M2\ell^2$ equal to $n+\frac12$ gives $\ell^2=(2n+1)/M$.
\item If one imposes $E_{\rm YM}=3\pi^2/g^2=n+\frac12$, then $\ell^2=4\pi G/g^2=\frac{4G}{3\pi}(n+\frac12)$ and $g^2/(4\pi)=3\pi/(4n+2)$.
\item The weak-coupling Yang--Mills levels $2m+1$ never equal $n+\frac12$.
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{(i)} $\psi'=-M\Omega\psi$ and $\psi''=(M^2\Omega^2-M)\psi$. So
\[
-\frac{\psi''}{2M}+\frac M2\Omega^2\psi=\Bigl(\frac12-\frac{M\Omega^2}2+\frac{M\Omega^2}2\Bigr)\psi=\frac12\psi .
\]
$H_\Omega$ is minus an oscillator, so its levels are the negatives of the usual ones \ext.

\emph{(ii)} Solve $\frac M2\ell^2=n+\frac12$ for $\ell^2$.

\emph{(iii)} $1/g^2=(n+\frac12)/(3\pi^2)$. Then $\ell^2=4\pi G(n+\frac12)/(3\pi^2)$ and $g^2/(4\pi)=3\pi^2/(4\pi(n+\frac12))$.

\emph{(iv)} $2m+1$ is an integer and $n+\frac12$ is not.

\emph{The number in the text.} $4n+2=3\pi\cdot137.036\approx1291.5$, so $n\approx322.38$.
\end{proof}


We stress what (iii) is and is not. It is the formal consequence of identifying the classical Yang--Mills energy with a level of the gravitational oscillator. It is \emph{not} a derivation of a coupling constant: the identification is not justified by a consistent quantisation of the constrained system; part (iv) shows that the two natural level sets do not even coincide; and $g$ is the $\SU(2)$ coupling of this model, not the electromagnetic coupling. If one nevertheless inserted the measured value $1/137.036$ for $g^2/4\pi$, one would need $n\approx322.38$, which is not an integer. We record the condition only as an example of the kind of statement the formalism produces, and as a warning \nogo.
