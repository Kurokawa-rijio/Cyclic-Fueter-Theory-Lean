\chapter{Further Results and Open Problems}\markboth{\MakeUppercase{Chapter 31.\ Further results, open problems}}{}
\label{ch:open}

\begin{usesbox}
\textbf{This chapter uses:} the whole book.\\
\textbf{It provides:} several results that sharpen earlier chapters (zero total energy of closed slices; why one conformal factor cannot interpolate between the two pieces; the gauge and gravitational-wave modes of the Tolman universe; the sphaleron direction; the Prasad--Sommerfield monopole as the statistics of an orientation; an alternative collapse bound; the inversion symmetry of the action; the shape modes of the internal $S^9$ and the origin of an adjoint Higgs field); and a list of open problems, each stated with its present status.
\end{usesbox}

This last chapter has two purposes. The first is to collect results that complete arguments made earlier but did not fit the logical line of the chapters where they belong. The second is to state clearly what has not been done. The book has proved many things about a specific configuration, a closed Tolman universe glued to a Bertotti--Robinson throat, and about the geometry of states that runs through it. It has not explained why nature should realise this configuration, and it has not derived any constant of nature.

\section{Zero total energy of a closed slice}

\begin{proposition}\label{prop:zeroenergy}\ver\
Let $m=\frac R{2G}(1-R'^2)$ be the Misner--Sharp mass of a time-symmetric slice. Then $m'=\bigl(R'(1-R'^2)-2RR'R''\bigr)/(2G)$, and with the Hamiltonian constraint $2(1-R'^2-2RR'')/R^2=16\pi G\rho$ this is $m'=4\pi R^2\rho R'$. If the slice closes at both ends ($R(0)=R(L)=0$), then $\int_0^Lm'\,\dd x=m(L)-m(0)=0$. On the Tolman side $m=\ell\sin^3y/(2G)$, which is $0$ at the pole and $\ell/(2G)$ at the equator, the mass of the throat.
\end{proposition}

The total energy of a closed universe vanishes, as it must; the mass rises from the pole to the equator and is constant, $\ell/(2G)$, along the throat.

\section{Why one conformal factor cannot interpolate}

\begin{theorem}[Two values only]\label{thm:twovalues}\ver\nogo\
On a time-symmetric, conformally flat slice with $w=r'$, at an extremal sphere $r=\ell$, $r'=0$ at $\chi=\pi/2$:
\begin{enumerate}[label=(\roman*)]
\item $\dd r'/\dd\chi=(\Omega_{\chi\chi}-\Omega)/\Omega$, so $r''=(\Omega_{\chi\chi}-\Omega)/\Omega^2$; with the trace equation $-\Omega_{\eta\eta}+\Omega_{\chi\chi}=\Omega$ this is $r''=\Omega_{\eta\eta}/\Omega^2$;
\item the constraint $2\ell^2r''^2+2rr''+\ell^2(1-r'^2)^2/r^2-(1-r'^2)=0$ has only the roots $r''=0$ and $r''=-1/\ell$;
\item hence $\Omega_{\eta\eta}/\Omega^2\in\{0,-1/\ell\}$: within a single smooth conformal factor the Tolman value and the throat value cannot be joined continuously.
\end{enumerate}
Without the condition $w=r'$ the situation changes: the constraint at the extremal sphere gives $r''=(w^2-w^4/2-\ell^2w'^2)/\ell$, and for $w=0$ every value $r''\le0$ is attained by a suitable $w'$.
\end{theorem}

\begin{remark}\opn\
Is there a smooth spherically symmetric EYM solution, not satisfying $w=r'$, that interpolates between a Tolman-like region and a throat-like region? \Cref{thm:twovalues} shows that the band of allowed $r''$ is continuous once $w=r'$ is dropped, so the obstruction of \cref{ch:envelope} is specific to the spin-connection condition.
\end{remark}

\section{Modes of the Tolman universe}

\begin{proposition}[Gauge modes]\label{prop:gaugemodes}\ver\
The perturbations generated by $\partial_\eta$, $\xi_1=\sin\eta\cos\chi\,\partial_\eta+\cos\eta\sin\chi\,\partial_\chi$ and $\xi_2=\cos\eta\cos\chi\,\partial_\eta-\sin\eta\sin\chi\,\partial_\chi$ solve the linearised equations, with $(f,\delta w)=(-\sin\eta,0)$, $(\cos\chi\cos2\eta,-\cos\eta\sin^2\chi)$ and $(-\cos\chi\sin2\eta,\sin\eta\sin^2\chi)$. For a general perturbation, integrability in $\eta$ requires $L[f]=\sin\chi(f_{\eta\eta}+f)-3\cos\chi f_\chi=0$; together with the trace equation this gives $f_{\chi\chi}\sin\chi-f_\chi\cos\chi=0$. The amplitude mode $f=\cos\eta$ is excluded: the algebraically determined $\delta w=-\sin^2\chi/(2\cos\chi)$ has $\chi$-derivative $-\sin\chi-\sin^3\chi/(2\cos^2\chi)$, not the required $-\sin\chi$. The gauge perturbation $\cos\chi\cos2\eta$ gives $r=\frac12\sin2\chi\cos2\eta$, a $k=2$ mode; the three generators remove $2\times3=6=2+2+2$ phase-space dimensions, those of three oscillators.
\end{proposition}

\begin{proposition}[Reduced action and gravitational waves]\label{prop:gw}\ver\
$\sin^2\chi\,(6\Omega^2-6\Omega\hat\Box\Omega)=6r(r_{\eta\eta}-r_{\chi\chi})$ for $r=\Omega\sin\chi$. On the Tolman background ($a''=-a$), a tensor mode with $h''=-2(a'/a)h'-(n^2-1)h$ satisfies $(ah)''=-n^2(ah)$: gravitational waves oscillate with frequency $n$. The number of transverse traceless harmonics is $2(n+2)(n-2)=2(n^2-4)$, which is $10$ at the lowest level $n=3$.
\end{proposition}

\section{The sphaleron direction}

\begin{proposition}\label{prop:sphaleron}\ver\
For the homogeneous field $f''=2f(1-f^2)$, the energy $f'^2+(1-f^2)^2$ is conserved. At the top of the barrier, $\delta f''=2\delta f$, with growth rate $\sqrt2$. The geometry is unaffected: $\Omega=a\cos(\eta-\eta_0)$ has $\Omega'^2+\Omega^2=a^2$ independently of $f$. Under $\lambda\mapsto1-\lambda$, $f=2\lambda-1\mapsto-f$, and the potential $(1-f^2)^2$ is even, so the Tolman configuration $f=0$ is parity invariant.
\end{proposition}

So the Yang--Mills field of the Tolman universe sits on top of a barrier and can roll off in either direction with rate $\sqrt2$, while the scale factor continues to follow the Tolman law. Combined with the spectral flow of \cref{thm:s3dirac}, a roll-off is a fermion-number-changing event.

\section{The monopole core as the statistics of an orientation}

\begin{proposition}\label{prop:langevin}\ver\
\begin{enumerate}[label=(\roman*)]
\item $\frac12\int_{-1}^1e^{xu}\dd u=\sinh x/x=:Z(x)$, the partition function of a unit vector in a field.
\item With $w=1/Z=x/\sinh x$ and $h=Z'/Z=\coth x-1/x$ (the Langevin function), $w'=-wh$ holds identically, and $h'=(1-w^2)/x^2$ is equivalent to $ZZ''-Z'^2=(Z^2-1)/x^2$, which $Z=\sinh x/x$ satisfies.
\item These are the Bogomolny equations of the Prasad--Sommerfield monopole \cite{PrasadSommerfield1975}: $w=1$, $h=0$ at the centre and $w\to0$, $h\to1$ at infinity.
\end{enumerate}
\end{proposition}

At the centre the monopole looks like the pole of the Tolman sphere ($w=1$, symmetric phase); far away it looks like the throat ($w=0$, $h=v$). The Higgs field is the mean orientation of a classical spin in a field, and the gauge field is the inverse partition function.

\begin{proposition}[Higgs fields break the Tolman solution]\label{prop:higgsbreak}\ver\
In the conformal frame the Yang--Mills equation acquires the mass term $-\Omega^2m^2h^2w$ from an adjoint Higgs field, which is not conformally invariant. For $w=\cos\chi$ the residual is $-\Omega^2m^2h^2\cos\chi\ne0$: with a massive Higgs field the Tolman configuration is no longer a solution.
\end{proposition}

\begin{proposition}[Dyons]\label{prop:dyonbf}\ver\
In a uniform electric field in $\mathrm{AdS}_2$ the effective mass of a charged field is $m^2\ell^2-(qE\ell^2)^2$ \ext. For the throat's mode, $-1-x^2<-\frac14$ for every $x$: an electric field makes the BF violation worse.
\end{proposition}

\begin{proposition}[A Hawking--Page transition]\label{prop:hp}\ver\
$\frac r{4G}(1-r^2/L^2)<0$ iff $r>L$, and the temperature $T=(1+3r^2/L^2)/(4\pi r)$ equals $1/(\pi L)$ at $r=L$ (formulae from \cite{HawkingPage1983} \ext).
\end{proposition}

\section{An alternative collapse bound}

\begin{proposition}\label{prop:altcollapse}\ver\
\begin{enumerate}[label=(\roman*)]
\item The partial derivatives of the throat Hamiltonian $H=\frac1{4c}(ap_a^2/b^2-2p_ap_b/b)-ca(1-\ell^2/b^2)$ are explicit, and $S=a\sigma(b)$ with $\sigma^2=Cb-4c^2(b^2+\ell^2)$, $\sigma\sigma'=(C-8c^2b)/2$ solves $H=0$ exactly.
\item At the reflection-symmetric centre of the double-null metric ($r_u=r_v$, $\sigma_u=\sigma_v$, $w_v=-w_u$), the double-null equations with the constraint reduce to the central equation \eqref{eq:centre}.
\item With lapse $\operatorname{sech}(t/\ell)$, the arrival time of a front is $\int_0^z\dd t/\cosh(t/\ell)=\ell\gd(z/\ell)$.
\item (Energy inequality) From a turning point, during collapse with $W^2r^2\ge\kappa$: $r(1+\dot r^2)+\ell^2/r-2\kappa/r\ge r_1+\ell^2/r_1-2\kappa/r_1$, hence $r\dot r^2\ge(r_1-r)+(2\kappa-\ell^2)(1/r-1/r_1)$ and, for $\kappa\ge\ell^2/2$, $\dot r^2\ge(\kappa-\ell^2/2)(1/r^2-1/r_1^2)$; the integrand $-2W^2\dot r+2\kappa\dot r/r^2$ is non-negative.
\item (Comparison) If $\dot r\le-\sqrt{A(1/r^2-1/r_1^2)}$ with $A>0$ on $[\tau_0,\tau_1]$, then $\tau_1-\tau_0\le r_1^2/\sqrt A$. With $A=\kappa-\ell^2/2>0$ the collapse time is at most $r_1^2/\sqrt{\kappa-\ell^2/2}$.
\end{enumerate}
\end{proposition}

This bound is weaker than \cref{thm:centrecollapse} near the threshold but independent of it; part (ii) justifies the central equation used in \cref{ch:collapse}.

\section{Inversion and the action}

\begin{proposition}\label{prop:inversion}\ver\
Under $r\mapsto\ell^2/r$, $a\mapsto ar/\ell$, the Kantowski--Sachs kinetic term is multiplied by $-\ell^3/r^3$ and the potential by $-r^3/\ell^3$. After a change of lapse the action changes only by a sign: solutions are mapped to solutions.
\end{proposition}

\section{Shape modes of the internal sphere and an adjoint Higgs field}

\begin{proposition}\label{prop:s9shape}\ver\
\begin{enumerate}[label=(\roman*)]
\item The second variation of a conformal deformation by a degree-$k$ harmonic on $S^n$ has coefficient $k(k+n-1)-n=(k-1)(k+n)$: $-n$ for $k=0$, $0$ for $k=1$ (conformal Killing directions), $n+2$ for $k=2$. On $S^9$ the $k=2$ coefficient is $11$ and there are $10\cdot11/2-1=54$ such harmonics.
\item $54=1+9+20+24$ and $45=6+15+24$ (dimension counts of the decompositions under $SO(4)\times SO(6)$), and $9=1+3+5$ for the $(3,3)$ component \ext.
\end{enumerate}
\end{proposition}

The ellipsoidal deformations of \cref{ch:intA} form the symmetric traceless $\mathbf{54}$ of $SO(10)$; under Pati--Salam it contains a $(3,3)$, which under the diagonal $SU(2)$ contains an adjoint $\mathbf3$. This is a candidate origin for the adjoint Higgs field of \cref{ch:higgs}, recorded here as an observation.

\begin{proposition}[Backreaction is unavoidable]\label{prop:areaQ}\ver\
The area $R=r^2$ satisfies $R_{uv}=(\Omega/r^2)(\ell^2(1-w^2)^2-r^2)$, and $Q=\log(\Omega/r^2)$ satisfies $Q_{uv}=6r_ur_v/r^2-3\Omega\ell^2(1-w^2)^2/r^4+2\Omega/r^2$. The derivative of the source with respect to $w$ is $12\Omega\ell^2w(1-w^2)/r^4$, non-zero whenever $w\ne0$ and $w^2\ne1$: a Yang--Mills perturbation backreacts on the geometry at first order.
\end{proposition}

\section{Open problems}

We close with the questions that this book leaves open. Each is stated with its present status.

\begin{enumerate}[label=\arabic*.]
\item \textbf{The fine-structure constant.} \opn\ Nothing in this book derives the fine-structure constant, or any other dimensionless coupling, from first principles. The scale $\ell=\sqrt{4\pi G}/g$ is fixed by the coupling, not the other way round. The level conditions of \cref{prop:levels,prop:confframe} are formal and, as shown there, inconsistent with the measured value if taken literally.
\item \textbf{Why glue?} \opn\ \Cref{thm:joinunique} shows that if the Tolman universe and the throat are joined, the join is unique. Nothing shows that they must be joined. A dynamical or quantum selection mechanism is missing.
\item \textbf{Global existence and nonlinear stability.} \opn\ Local existence for the glued data rests on external theorems with low regularity (\cref{ch:evolution}). The long-throat instability (\cref{thm:longthroat}) and the collapse of the central sphere (\cref{thm:centrecollapse}) are proved, but the global structure of the evolved spacetime, including the nature of its singularities, is not known.
\item \textbf{Beyond spherical symmetry.} \opn\ \Cref{ch:shape} identified the shape degrees of freedom and showed that they focus. A full analysis without symmetry, including the coupling of gravitational waves to the Yang--Mills instability, remains to be done.
\item \textbf{Quantum theory.} \opn\ The Wheeler--DeWitt analysis of \cref{ch:wdw} concerns the throat minisuperspace. A quantisation of the full spherically symmetric system, with the junction treated quantum mechanically, is not available.
\item \textbf{Internal symmetry.} \opn\ The octonionic constructions of \cref{ch:intA} and \cref{ch:intB} give an explicit $\Spin(10)$ Weyl block and a Pati--Salam split from the slice, but no dynamical mechanism selects one generation, three generations, or the observed masses.
\item \textbf{The global minimum on ellipsoids.} \opn\ Whether the round $S^9$ minimises the normalised lowest Dirac eigenvalue among all ellipsoids (\cref{ch:intA}) is open; only the local statement is proved.
\end{enumerate}

\begin{litbox}
The Prasad--Sommerfield monopole is from \cite{PrasadSommerfield1975}; the Langevin function is classical; the other external inputs are those cited in the earlier chapters. New in this chapter are the two-value theorem for a single conformal factor (\cref{thm:twovalues}), the identification of the Prasad--Sommerfield profile with the statistics of an orientation (\cref{prop:langevin}), the gauge-mode analysis of the Tolman universe (\cref{prop:gaugemodes}), the alternative collapse bound (\cref{prop:altcollapse}) and the inversion symmetry of the Kantowski--Sachs action (\cref{prop:inversion}).
\end{litbox}

\end{exercise}
\begin{exercise}

\end{exercise}

\end{exercise}
